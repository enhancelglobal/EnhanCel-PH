function askCoachPreset(q){appendMsg('user',q); setTimeout(()=>appendMsg('ai',coachReply(q)),140);}
document.getElementById('coachInput').addEventListener('keydown', function(e){ if(e.key==='Enter'&&!e.shiftKey){ e.preventDefault(); sendCoach(); }});
function calculateIncome(){
  const referralPct=Number(document.getElementById('myPackage').value||0);
  const packageEV=Number(document.getElementById('downlineEV').value||0);
  const evRate=13;
  const count=Number(document.getElementById('downlineCount').value||0);
  const referralPerPackage=packageEV*referralPct/100*evRate;
  const totalReferral=referralPerPackage*count;
  const result=document.getElementById('calcResult'); result.dataset.calculated='1';

Pairing, matching, overriding, and leadership bonus depend on structure and qualification.</div>`;
}
